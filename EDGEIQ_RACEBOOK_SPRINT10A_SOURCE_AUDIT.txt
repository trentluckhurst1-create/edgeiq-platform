0003: function clean(value: unknown): string {
0004:   return String(value ?? "").trim();
0005: }

0007: function scoreTextMatch(a: unknown, b: unknown, weight: number): number {
0008:   const left = clean(a).toUpperCase();
0009:   const right = clean(b).toUpperCase();
0010:   if (!left || !right) return weight * 0.45;
0011:   if (left === right) return weight;
0012:   if (left.includes(right) || right.includes(left)) return weight * 0.8;
0013:   return weight * 0.35;
0014: }

0016: function scoreDistance(a: unknown, b: unknown, weight: number): number {
0017:   const n1 = Number(clean(a).replace(/[^\d.]/g, ""));
0018:   const n2 = Number(clean(b).replace(/[^\d.]/g, ""));
0019:   if (!Number.isFinite(n1) || !Number.isFinite(n2)) return weight * 0.45;
0020:   const diff = Math.abs(n1 - n2);
0021:   if (diff <= 50) return weight;
0022:   if (diff <= 100) return weight * 0.85;
0023:   if (diff <= 200) return weight * 0.65;
0024:   return weight * 0.35;
0025: }
